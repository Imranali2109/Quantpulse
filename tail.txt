   317	                  </div>
   318	                </div>
   319	              )}
   320	            </Pane>
   321	          </div>
   322	
   323	          {/* Volume & Volatility */}
   324	          <div className="col-span-1 flex flex-col">
   325	            <Pane title="Volume & Risk" loading={analysisLoading} error={analysisError}>
   326	              {analysis && (

   328	                  <div className="flex items-center justify-between">
   329	                    <span className="text-xs text-zinc-400">VWAP</span>
   330	                    <span className="text-xs font-semibold text-zinc-300">{quote ? formatCurrency(analysis.indicators.volume.vwap, quote.currency) : analysis.indicators.volume.vwap.toFixed(2)}</span>
   331	                  </div>

   352	
   353	          {/* News & Accuracy */}
   354	          <div className="col-span-1 lg:col-span-1 flex flex-col gap-4">
   355	            <Pane title="Model Accuracy" loading={accuracyLoading}>
   356	              {accuracy && (

   365	                  </div>

   367	                </div>
   368	              )}
   369	            </Pane>
   370	             <Pane title="News & Sentiment" loading={newsLoading}>
   371	              {news && news.length > 0 ? (
   372	                <div className="space-y-3">

   386	            </Pane>
   387	          </div>
   388	
   389	        </div>
   390	      </main>
   391	    </div>
